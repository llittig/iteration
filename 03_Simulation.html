Let’s do this but keep track of results

output = vector("list", length = 1000)

for (i in 1:1000) {
  
  output[[i]] = sim_mean_sd(n_obs = 30)
  
}

inspect results

output |> 
  bind_rows() |> 
  summarize(
    emp_m = mean(mean),
    emp_se = sd(mean)
  )
## # A tibble: 1 × 2

## 1  12.0  0.553
output |> 
  bind_rows() |> 
  ggplot(aes(x = mean)) +
  geom_density()

Let’s do this using tibbles and map instead!

sim_results_df =
  tibble(
    samp_size = rep(30, 1000)
  ) |> 
  mutate(
    est_df = map(samp_size, sim_mean_sd)
  ) |> 
  unnest(est_df)

Look at results

sim_results_df |> 
  ggplot(aes(x = mean)) +
  geom_density()

change n

sim_results_df =
  expand_grid(
    samp_size = c(30, 60, 120, 240),
    iter = 1:1000
  ) |> 
   mutate(
    est_df = map(samp_size, sim_mean_sd)
  ) |> 
  unnest(est_df)
sim_results_df |> 
  mutate(
    samp_size = str_c("n = ", samp_size),
    samp_size = fct_inorder(samp_size)
  ) |> 
  ggplot(aes(x = samp_size, y = mean)) +
  geom_violin()

Look at the numbers

sim_results_df |> 
  mutate(
    samp_size = str_c("n = ", samp_size),
    samp_size = fct_inorder(samp_size)
  ) |> 
  group_by(samp_size) |> 
  summarize(
    emp_m = mean(mean),
    emp_se = sd(mean)
  )
